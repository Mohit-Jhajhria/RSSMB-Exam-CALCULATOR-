<script>
document.getElementById('calculateBtn').addEventListener('click', function() {
// Hide previous results and errors
document.getElementById('results').classList.add('hidden');
const errorEl = document.getElementById('error');
errorEl.classList.add('hidden');
errorEl.textContent = '';

try {
// Get Paper 1 values
const p1_maxMarks = parseFloat(document.getElementById('p1_maxMarks').value) || 0;
const p1_totalQ = parseFloat(document.getElementById('p1_totalQ').value) || 0;
const p1_totalCorrect = parseFloat(document.getElementById('p1_totalCorrect').value) || 0;
const p1_totalWrong = parseFloat(document.getElementById('p1_totalWrong').value) || 0;
const p1_delRight = parseFloat(document.getElementById('p1_delRight').value) || 0;
const p1_delWrong = parseFloat(document.getElementById('p1_delWrong').value) || 0;

// Get Paper 2 values
const p2_maxMarks = parseFloat(document.getElementById('p2_maxMarks').value) || 0;
const p2_totalQ = parseFloat(document.getElementById('p2_totalQ').value) || 0;
const p2_totalCorrect = parseFloat(document.getElementById('p2_totalCorrect').value) || 0;
const p2_totalWrong = parseFloat(document.getElementById('p2_totalWrong').value) || 0;
const p2_delRight = parseFloat(document.getElementById('p2_delRight').value) || 0;
const p2_delWrong = parseFloat(document.getElementById('p2_delWrong').value) || 0;

// Helper function to calculate one paper
function calculatePaper(maxMarks, totalQ, totalCorrect, totalWrong, delRight, delWrong) {
if (totalQ <= 0) {
throw new Error('Total Questions must be greater than 0.');
}
// Input validation checks
if (totalCorrect < 0 || totalWrong < 0 || delRight < 0 || delWrong < 0) {
throw new Error('All answer counts must be zero or positive.');
}
if (totalCorrect < delRight) {
throw new Error('"Deleted Right" cannot be more than "Total Correct" answers for this paper.');
}
if (totalWrong < delWrong) {
throw new Error('"Deleted Wrong" cannot be more than "Total Wrong" answers for this paper.');
}
// Check if attempted questions is reasonable
if ((totalCorrect + totalWrong + totalQ - totalCorrect - totalWrong) > totalQ) {
// This ensures total attempted (Correct + Wrong + Unattempted) doesn't exceed Total Qs
}

// This is the total number of questions *officially* deleted by the exam board
const totalDeleted = delRight + delWrong;
const validQuestions = totalQ - totalDeleted;

if (validQuestions <= 0) {
throw new Error('Number of Valid Questions is 0 or less. Check deleted questions count.');
}

// Key Factor (KF)
// KF = Max Marks / Valid Questions
const kf = maxMarks / validQuestions;
// The user's final RQ and WQ are their *remaining* correct and wrong answers.
const finalRQ = totalCorrect - delRight;
const finalWQ = totalWrong - delWrong;
// The formula from the image is:
// Total Marks = (RQ * KF) - (WQ * KF) / 3
const attemptedMarks = finalRQ * kf;
const negativeMarks = (finalWQ * kf) / 3;
const paperScore = attemptedMarks - negativeMarks;

return { paperScore, totalDeleted, validQuestions, kf, finalRQ, finalWQ };
}

// Calculate scores
const p1_results = calculatePaper(p1_maxMarks, p1_totalQ, p1_totalCorrect, p1_totalWrong, p1_delRight, p1_delWrong);
const p2_results = calculatePaper(p2_maxMarks, p2_totalQ, p2_totalCorrect, p2_totalWrong, p2_delRight, p2_delWrong);

// Display Paper 1 Results
document.getElementById('p1_res_deleted').textContent = p1_results.totalDeleted;
document.getElementById('p1_res_valid').textContent = p1_results.validQuestions;
document.getElementById('p1_res_kf').textContent = p1_results.kf.toFixed(4); // Show 4 decimal places for kf
document.getElementById('p1_res_rq').textContent = p1_results.finalRQ;
document.getElementById('p1_res_wq').textContent = p1_results.finalWQ;
document.getElementById('p1_res_score').textContent = p1_results.paperScore.toFixed(3);

// Display Paper 2 Results
document.getElementById('p2_res_deleted').textContent = p2_results.totalDeleted;
document.getElementById('p2_res_valid').textContent = p2_results.validQuestions;
document.getElementById('p2_res_kf').textContent = p2_results.kf.toFixed(4);
document.getElementById('p2_res_rq').textContent = p2_results.finalRQ;
document.getElementById('p2_res_wq').textContent = p2_results.finalWQ;
document.getElementById('p2_res_score').textContent = p2_results.paperScore.toFixed(3);

// Display Total Score
const totalScore = p1_results.paperScore + p2_results.paperScore;
document.getElementById('total_score').textContent = totalScore.toFixed(3);

// Show the results block
document.getElementById('results').classList.remove('hidden');

} catch (e) {
// Display error message
errorEl.textContent = `Error: ${e.message}`;
errorEl.classList.remove('hidden');
}
});
</script>
